MindkeeperMap in action

A live mindmap rendered with mindkeeper-map.js. Drag nodes to reposition. Scroll to zoom. Pan by dragging the background.

Usage — mindkeeper-map.js 
// 1. Include the script
<script src="mindkeeper-map.js"></script>

// 2. Create a container
<div id="container" style="width:100%;height:500px;"></div>

// 3. Initialize with tree data
const map = new MindkeeperMap('#container');
map.init({
  topic: 'Root',
  children: [
    { topic: 'Branch A', children: [{ topic: 'Leaf' }] },
    { topic: 'Branch B' },
  ]
});

// 4. Export
map.exportPNG('mindmap.png');
map.exportSVG('mindmap.svg');
